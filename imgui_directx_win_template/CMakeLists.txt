cmake_minimum_required(VERSION 3.25)
project(imgui_directx_win_template LANGUAGES C CXX)

if(NOT WIN32)
    message(FATAL_ERROR "This template is Windows-only. Use the GLFW/OpenGL template for cross-platform builds.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

set(IMGUI_DIRECTX_VERSION "11" CACHE STRING "Dear ImGui DirectX backend to build: 9, 10, 11, or 12")
set_property(CACHE IMGUI_DIRECTX_VERSION PROPERTY STRINGS 9 10 11 12)

if(NOT IMGUI_DIRECTX_VERSION MATCHES "^(9|10|11|12)$")
    message(FATAL_ERROR "IMGUI_DIRECTX_VERSION must be one of: 9, 10, 11, 12")
endif()

message(STATUS "")
message(STATUS "==== Compiler Information ====")
message(STATUS "  C++ compiler      : ${CMAKE_CXX_COMPILER}")
message(STATUS "  C++ standard      : ${CMAKE_CXX_STANDARD}")
message(STATUS "  Compiler ID       : ${CMAKE_CXX_COMPILER_ID}")
message(STATUS "  Compiler version  : ${CMAKE_CXX_COMPILER_VERSION}")
#message(STATUS "  Compile features  : ${CMAKE_CXX_COMPILE_FEATURES}")
message(STATUS "")
message(STATUS "  C compiler        : ${CMAKE_C_COMPILER}")
message(STATUS "  C standard        : ${CMAKE_C_STANDARD}")
message(STATUS "  Compiler ID       : ${CMAKE_C_COMPILER_ID}")
message(STATUS "  Compiler version  : ${CMAKE_C_COMPILER_VERSION}")
#message(STATUS "  Compile features  : ${CMAKE_C_COMPILE_FEATURES}")
message(STATUS "")
message(STATUS "  DirectX backend   : DX${IMGUI_DIRECTX_VERSION}")
message(STATUS "===============================")
message(STATUS "")

add_subdirectory(libs/imgui)
add_subdirectory(app)
